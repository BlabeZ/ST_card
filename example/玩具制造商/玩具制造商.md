<script>
    (function() {
        const dashboard = document.getElementById('main-dashboard');
        if (!dashboard || dashboard.dataset.initialized) return;
        dashboard.dataset.initialized = 'true';

        const statusTextPanel = document.getElementById('status-text-panel');
        if (!statusTextPanel) return;

        const rawText = statusTextPanel.innerHTML;

        // 1. 解析BPM和ECG数据
        const bpmMatch = rawText.match(/心率BPM:\s*(\d+)/);
        const ecgMatch = rawText.match(/波形ECG:\s*([\d,.\s-]+)/);

        const bpm = bpmMatch ? parseInt(bpmMatch[1], 10) : 0;
        const ecgPattern = ecgMatch ? ecgMatch[1].split(',').map(p => parseFloat(p.trim())) : [];

        // 2. 更新BPM显示
        const bpmValueEl = document.getElementById('bpm-value');
        if (bpmValueEl && bpm > 0) {
            bpmValueEl.textContent = bpm;
        }

        // 3. 强化ECG波形图生成
        const svgPath = document.getElementById('ecg-path-line');
        if (svgPath && ecgPattern.length > 1 && bpm > 0) {
            const svgWidth = 400;
            const svgHeight = 100;

            // 计算一个心跳周期需要多少个基础模式点
            const pointsPerBeat = ecgPattern.length;
            // 计算一个心跳在SVG中的宽度
            const beatWidth = (60 / bpm) * 30; // 假设每秒滚动30个SVG单位

            // 计算屏幕上需要多少个心跳来填满并循环
            const beatsToFill = Math.ceil(svgWidth / beatWidth) + 2; // +2确保循环无缝

            let fullPathData = '';
            let currentX = 0;

            // 生成足够长的路径数据
            for (let i = 0; i < beatsToFill; i++) {
                const stepX = beatWidth / (pointsPerBeat - 1);
                for (let j = 0; j < pointsPerBeat; j++) {
                    const x = currentX + j * stepX;
                    const y = svgHeight - ecgPattern[j]; // Y轴反转
                    if (i === 0 && j === 0) {
                        fullPathData += `M ${x},${y}`;
                    } else {
                        fullPathData += ` L ${x},${y}`;
                    }
                }
                currentX += beatWidth;
            }

            svgPath.setAttribute('d', fullPathData);

            // 设置滚动动画
            // pathLength近似为生成的总宽度
            const pathLength = currentX;
            svgPath.style.strokeDasharray = `${pathLength}`;
            svgPath.style.strokeDashoffset = `${pathLength}`;
            svgPath.style.animation = `draw-ecg ${beatsToFill * (60 / bpm)}s linear infinite`;

            // 重新定义动画以匹配动态计算的偏移量
            const styleSheet = document.styleSheets[0];
            const keyframes = `@keyframes draw-ecg { to { stroke-dashoffset: -${pathLength}; } }`;

            // 确保动画规则只被插入一次
            if (!dashboard.dataset.animationDefined) {
                 styleSheet.insertRule(keyframes, styleSheet.cssRules.length);
                 dashboard.dataset.animationDefined = 'true';
            }
        }

        // 4. 清理文本面板
        statusTextPanel.innerHTML = rawText
            .replace(/-\s*心率BPM:[\s\S]*?(\r\n|\n|$)/, '')
            .replace(/-\s*波形ECG:[\s\S]*?(\r\n|\n|$)/, '')
            .trim();

    })();
</script>
